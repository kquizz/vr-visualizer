**scripts/autoloads/shader_bridge.gd** - Replace placeholder with full implementation:
```gdscript
extends Node

## Stem prefixes matching AudioManager.STEM_NAMES order
const STEM_PREFIXES: PackedStringArray = ["drums", "bass", "vocals", "other"]

func _ready() -> void:
    # Register all global shader parameters
    # These MUST be registered before any shader tries to read them
    for prefix in STEM_PREFIXES:
        RenderingServer.global_shader_parameter_add(
            prefix + "_energy", RenderingServer.GLOBAL_VAR_TYPE_FLOAT, 0.0
        )
        RenderingServer.global_shader_parameter_add(
            prefix + "_peak_freq", RenderingServer.GLOBAL_VAR_TYPE_FLOAT, 0.0
        )
        RenderingServer.global_shader_parameter_add(
            prefix + "_bands_low", RenderingServer.GLOBAL_VAR_TYPE_VEC4, Vector4.ZERO
        )
        RenderingServer.global_shader_parameter_add(
            prefix + "_bands_high", RenderingServer.GLOBAL_VAR_TYPE_VEC4, Vector4.ZERO
        )

func _process(_delta: float) -> void:
    var data: Array[AudioData] = AudioManager.stem_data
    for i in range(min(data.size(), STEM_PREFIXES.size())):
        var prefix: String = STEM_PREFIXES[i]
        RenderingServer.global_shader_parameter_set(
            prefix + "_energy", data[i].energy
        )
        RenderingServer.global_shader_parameter_set(
            prefix + "_peak_freq", data[i].peak_frequency
        )
        RenderingServer.global_shader_parameter_set(
            prefix + "_bands_low",
            Vector4(
                data[i].bands[0], data[i].bands[1],
                data[i].bands[2], data[i].bands[3]
            )
        )
        RenderingServer.global_shader_parameter_set(
            prefix + "_bands_high",
            Vector4(
                data[i].bands[4], data[i].bands[5],
                data[i].bands[6], 0.0
            )
        )
```

This pushes 16 global uniforms per frame (4 per stem: energy, peak_freq, bands_low, bands_high). Uses RenderingServer.global_shader_parameter_add in _ready to register them, then global_shader_parameter_set in _process to update values. All shaders anywhere in the project can read these with `global uniform`.

**shaders/test_reactive.gdshader** - Test shader proving audio-to-visual pipeline:
```gdshader
shader_type spatial;
render_mode unshaded;

global uniform float drums_energy;
global uniform float bass_energy;
global uniform float vocals_energy;
global uniform float other_energy;
global uniform vec4 drums_bands_low;

void fragment() {
    // Color channels driven by stem energy:
    // Red = drums, Green = bass, Blue = vocals
    vec3 color = vec3(drums_energy, bass_energy, vocals_energy);

    // Add some variation from drum sub-bass band
    float pulse = drums_bands_low.x * 0.5;

    ALBEDO = color + vec3(pulse);
    EMISSION = color * 2.0 + vec3(pulse) * 3.0;
}

void vertex() {
    // Scale mesh based on combined energy
    float scale_factor = 1.0 + (drums_energy + bass_energy) * 0.3;
    VERTEX *= scale_factor;
}
```

This shader uses ONLY global uniforms (no per-material set_shader_parameter). It visually reacts to audio: color shifts with stem energies, mesh pulses with drums+bass. Render mode is unshaded so emission is directly visible without lighting.

**scripts/debug_overlay.gd** - FFT debug display using Label3D nodes:
```gdscript
extends Node3D

var _labels: Array[Label3D] = []
const STEM_NAMES: PackedStringArray = ["Drums", "Bass", "Vocals", "Other"]
const STEM_COLORS: Array[Color] = [
    Color(1.0, 0.3, 0.3),  # Drums: red
    Color(0.3, 0.3, 1.0),  # Bass: blue
    Color(0.3, 1.0, 0.3),  # Vocals: green
    Color(1.0, 1.0, 0.3),  # Other: yellow
]

func _ready() -> void:
    for i in range(4):
        var label := Label3D.new()
        label.position = Vector3(-1.5 + i * 1.0, 2.5, -3.0)
        label.font_size = 24
        label.modulate = STEM_COLORS[i]
        label.billboard = BaseMaterial3D.BILLBOARD_ENABLED
        label.no_depth_test = true
        label.outline_size = 8
        add_child(label)
        _labels.append(label)

func _process(_delta: float) -> void:
    var data: Array[AudioData] = AudioManager.stem_data
    for i in range(min(data.size(), _labels.size())):
        _labels[i].text = "%s\nE: %.2f\nPk: %.0f Hz\n%s" % [
            STEM_NAMES[i],
            data[i].energy,
            data[i].peak_frequency,
            _format_bands(data[i].bands)
        ]

func _format_bands(bands: Array[float]) -> String:
    var result: String = ""
    for b in bands:
        var filled: int = int(b * 5)
        result += "|" + "#".repeat(filled) + ".".repeat(5 - filled) + "|\n"
    return result
```

Debug overlay uses Label3D with billboard mode so text always faces the camera (works in both VR and flat-screen). Positioned at y=2.5 z=-3.0 (above eye level, 3 meters in front). Shows energy, peak frequency, and ASCII bar graphs for each of the 7 bands per stem.

**scripts/stem_loader.gd** - Auto-loads and plays test stems:
```gdscript
extends Node

## Paths to test stem files (OGG format)
@export var drums_path: String = "res://audio/stems/test_drums.ogg"
@export var bass_path: String = "res://audio/stems/test_bass.ogg"
@export var vocals_path: String = "res://audio/stems/test_vocals.ogg"
@export var other_path: String = "res://audio/stems/test_other.ogg"

func _ready() -> void:
    # Wait one frame for AudioManager to initialize
    await get_tree().process_frame
    await get_tree().process_frame

    # Check if stem files exist before playing
    if not FileAccess.file_exists(drums_path):
        push_warning("StemLoader: No test stems found at %s. Place OGG stems in audio/stems/ to hear audio." % drums_path)
        print("StemLoader: To prepare test stems, run:")
        print("  demucs -n htdemucs your_track.mp3")
        print("  ffmpeg -i separated/htdemucs/your_track/drums.wav -c:a libvorbis -q:a 5 audio/stems/test_drums.ogg")
        return

    AudioManager.play_stems(drums_path, bass_path, vocals_path, other_path)
    print("StemLoader: Playing test stems")
```
