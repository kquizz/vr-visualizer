    ```gdscript
    class_name MilkdropGridMesh
    extends RefCounted

    const GRID_X := 128
    const GRID_Y := 96

    var _mesh: ArrayMesh
    var _uvs: PackedVector2Array
    var _arrays: Array  # Cached arrays for surface update
    var _vertex_count: int  # (GRID_X + 1) * (GRID_Y + 1) = 12,417

    func create_mesh() -> ArrayMesh:
        # Create the static grid mesh topology (vertices + indices).
        # Vertices are positioned in normalized -1..1 space for rendering on a quad.
        # UVs start as identity (will be overwritten each frame).
        #
        # Vertex layout: (GRID_X+1) * (GRID_Y+1) = 129 * 97 = 12,513 vertices
        # Triangle count: GRID_X * GRID_Y * 2 = 24,576 triangles
        #
        _vertex_count = (GRID_X + 1) * (GRID_Y + 1)
        var verts := PackedVector3Array()
        verts.resize(_vertex_count)
        _uvs = PackedVector2Array()
        _uvs.resize(_vertex_count)
        var indices := PackedInt32Array()
        indices.resize(GRID_X * GRID_Y * 6)

        var v_idx := 0
        for gy in range(GRID_Y + 1):
            for gx in range(GRID_X + 1):
                var x := float(gx) / GRID_X
                var y := float(gy) / GRID_Y
                verts[v_idx] = Vector3(x * 2.0 - 1.0, y * 2.0 - 1.0, 0.0)
                _uvs[v_idx] = Vector2(x, y)
                v_idx += 1

        var i_idx := 0
        for gy in range(GRID_Y):
            for gx in range(GRID_X):
                var tl := gy * (GRID_X + 1) + gx
                var tr := tl + 1
                var bl := tl + (GRID_X + 1)
                var br := bl + 1
                indices[i_idx] = tl; i_idx += 1
                indices[i_idx] = bl; i_idx += 1
                indices[i_idx] = tr; i_idx += 1
                indices[i_idx] = tr; i_idx += 1
                indices[i_idx] = bl; i_idx += 1
                indices[i_idx] = br; i_idx += 1

        _arrays = []
        _arrays.resize(Mesh.ARRAY_MAX)
        _arrays[Mesh.ARRAY_VERTEX] = verts
        _arrays[Mesh.ARRAY_TEX_UV] = _uvs
        _arrays[Mesh.ARRAY_INDEX] = indices

        _mesh = ArrayMesh.new()
        _mesh.add_surface_from_arrays(Mesh.PRIMITIVE_TRIANGLES, _arrays)
        return _mesh
    ```

    **update_uvs() -- the core MilkDrop algorithm:**
    ```gdscript
    func update_uvs(preset: MilkdropPreset, per_frame_pool: Dictionary,
                     nseel: NseelInterpreter, time: float) -> void:
        # Compute per-vertex UVs using MilkDrop's 7-step transform chain.
        # If preset has no per_pixel_ast, use per-frame values directly (optimization).

        var has_per_pixel := preset.per_pixel_ast != null

        # Read per-frame warp parameters from pool (these were just computed by per-frame equations)
        var pf_zoom: float = per_frame_pool.get("zoom", preset.zoom)
        var pf_zoomexp: float = per_frame_pool.get("zoomexp", preset.zoomexp)
        var pf_rot: float = per_frame_pool.get("rot", preset.rot)
        var pf_warp: float = per_frame_pool.get("warp", preset.warp)
        var pf_cx: float = per_frame_pool.get("cx", preset.cx)
        var pf_cy: float = per_frame_pool.get("cy", preset.cy)
        var pf_dx: float = per_frame_pool.get("dx", preset.dx)
        var pf_dy: float = per_frame_pool.get("dy", preset.dy)
        var pf_sx: float = per_frame_pool.get("sx", preset.sx)
        var pf_sy: float = per_frame_pool.get("sy", preset.sy)

        var warp_time: float = time * preset.warp_anim_speed
        var warp_scale_inv: float = 1.0 / maxf(preset.warp_scale, 0.001)

        var v_idx := 0
        for gy in range(GRID_Y + 1):
            for gx in range(GRID_X + 1):
                var x := float(gx) / GRID_X
                var y := float(gy) / GRID_Y

                # Start with per-frame values
                var zoom := pf_zoom
                var zoomexp := pf_zoomexp
                var rot := pf_rot
                var warp := pf_warp
                var cx := pf_cx; var cy := pf_cy
                var dx := pf_dx; var dy := pf_dy
                var sx := pf_sx; var sy := pf_sy

                # Per-vertex equations (if present)
                if has_per_pixel:
                    var rad := sqrt((x - 0.5) * (x - 0.5) + (y - 0.5) * (y - 0.5)) * 2.0
                    var ang := atan2(y - 0.5, x - 0.5)

                    # Create per-vertex pool: copy q1-q32 from per-frame, set vertex inputs
                    var pv_pool := {}
                    for qi in range(1, 33):
                        var qname := "q%d" % qi
                        if per_frame_pool.has(qname):
                            pv_pool[qname] = per_frame_pool[qname]
                    pv_pool["x"] = x
                    pv_pool["y"] = y
                    pv_pool["rad"] = rad
                    pv_pool["ang"] = ang
                    pv_pool["zoom"] = zoom
                    pv_pool["zoomexp"] = zoomexp
                    pv_pool["rot"] = rot
                    pv_pool["warp"] = warp
                    pv_pool["cx"] = cx; pv_pool["cy"] = cy
                    pv_pool["dx"] = dx; pv_pool["dy"] = dy
                    pv_pool["sx"] = sx; pv_pool["sy"] = sy

                    nseel.evaluate_compiled(preset.per_pixel_ast, pv_pool)

                    # Read back modified values
                    zoom = pv_pool.get("zoom", zoom)
                    zoomexp = pv_pool.get("zoomexp", zoomexp)
                    rot = pv_pool.get("rot", rot)
                    warp = pv_pool.get("warp", warp)
                    cx = pv_pool.get("cx", cx); cy = pv_pool.get("cy", cy)
                    dx = pv_pool.get("dx", dx); dy = pv_pool.get("dy", dy)
                    sx = pv_pool.get("sx", sx); sy = pv_pool.get("sy", sy)

                # 7-step UV transform chain (from MilkDrop3 source):
                var rad_for_zoom := sqrt((x - 0.5) * (x - 0.5) + (y - 0.5) * (y - 0.5)) * 2.0

                # Step 1: Zoom (radius-dependent via zoomexp)
                var zoom2 := pow(zoom, pow(zoomexp, rad_for_zoom * 2.0 - 1.0))
                var zoom2_inv := 1.0 / maxf(zoom2, 0.001)
                var u := (x - 0.5) * zoom2_inv + 0.5
                var v := (y - 0.5) * zoom2_inv + 0.5

                # Step 2: Stretch
                u = (u - cx) / maxf(sx, 0.001) + cx
                v = (v - cy) / maxf(sy, 0.001) + cy

                # Step 3: Warp (4 oscillating sine waves -- MilkDrop's organic flow)
                var f0 := 11.68 + 4.0 * cos(warp_time * 1.413 + 10.0)
                var f1 := 8.77 + 3.0 * cos(warp_time * 1.113 + 7.0)
                var f2 := 10.54 + 3.0 * cos(warp_time * 1.233 + 3.0)
                var f3 := 11.49 + 4.0 * cos(warp_time * 0.933 + 5.0)
                u += warp * 0.0035 * sin(warp_time * 0.333 + warp_scale_inv * (x * f0 - y * f3))
                v += warp * 0.0035 * cos(warp_time * 0.375 - warp_scale_inv * (x * f2 + y * f1))
                u += warp * 0.0035 * cos(warp_time * 0.753 - warp_scale_inv * (x * f1 - y * f2))
                v += warp * 0.0035 * sin(warp_time * 0.825 + warp_scale_inv * (x * f0 + y * f3))

                # Step 4: Rotation (around cx, cy)
                var u2 := u - cx
                var v2 := v - cy
                var cos_rot := cos(rot)
                var sin_rot := sin(rot)
                u = u2 * cos_rot - v2 * sin_rot + cx
                v = u2 * sin_rot + v2 * cos_rot + cy

                # Step 5: Translation
                u -= dx
                v -= dy

                # Steps 6-7: aspect ratio and texel offset (skip for square viewport)

                _uvs[v_idx] = Vector2(u, v)
                v_idx += 1

        # Update the mesh surface UVs in-place
        _arrays[Mesh.ARRAY_TEX_UV] = _uvs
        _mesh.clear_surfaces()
        _mesh.add_surface_from_arrays(Mesh.PRIMITIVE_TRIANGLES, _arrays)
    ```

    **shaders/milkdrop_warp.gdshader** -- Canvas item shader for the grid mesh inside the feedback SubViewport. The grid mesh provides warped UVs; this shader samples the previous frame at those UVs and applies decay.

    ```glsl
    shader_type canvas_item;

    uniform sampler2D prev_frame : filter_linear, repeat_enable;
    uniform float decay : hint_range(0.9, 1.0) = 0.98;
    uniform float gamma_adj : hint_range(0.5, 2.0) = 1.0;

    void fragment() {
        // UV comes from the grid mesh's warped coordinates
        vec3 color = texture(prev_frame, UV).rgb;

        // Apply decay (multiplicative fade)
        color *= decay;

        // Gamma adjustment
        color = pow(color, vec3(gamma_adj));

        COLOR = vec4(color, 1.0);
    }
    ```

    Key: This shader is SIMPLE because all the warp math happens on the CPU (per-vertex UV computation in grid_mesh.gd). The shader just samples the previous frame texture at the grid-provided UVs and applies decay. This is the MilkDrop approach -- the "warp" is in the mesh UVs, not in the shader.

    **shaders/milkdrop_display.gdshader** -- Spatial shader for the inverted sphere dome (same concept as existing warp_display.gdshader but for milkdrop).

    ```glsl
    shader_type spatial;
    render_mode unshaded, cull_back;

    uniform sampler2D milkdrop_texture : filter_linear;
    uniform float brightness : hint_range(0.5, 3.0) = 1.5;

    void fragment() {
        vec3 color = texture(milkdrop_texture, UV).rgb;
        color *= brightness;
        ALBEDO = color;
        EMISSION = color;
    }
    ```

  <action>
    Create `scripts/milkdrop/milkdrop_renderer.gd` -- the main orchestrator that runs the MilkDrop rendering pipeline each frame.

    This script is designed to be attached to the root Node3D of the milkdrop scene (created in Plan 06). It expects the scene tree to contain:
    - SubViewportA and SubViewportB (2048x2048, for ping-pong feedback)
    - GridMeshInstance (MeshInstance2D inside SubViewportA, uses milkdrop_warp.gdshader)
    - WarpDome (MeshInstance3D with inverted SphereMesh, uses milkdrop_display.gdshader)

    ```gdscript
    class_name MilkdropRenderer
    extends Node3D

    # Configuration
    const VIEWPORT_SIZE := 2048

    # Core components
    var _nseel := NseelInterpreter.new()
    var _audio_bridge := MilkdropAudioBridge.new()
    var _grid_mesh := MilkdropGridMesh.new()
    var _preset_loader := MilkdropPresetLoader.new()

    # Current state
    var _preset: MilkdropPreset = null
    var _per_frame_pool: Dictionary = {}
    var _time: float = 0.0
    var _frame_count: int = 0

    # Scene node references (set in _ready)
    var _viewport_a: SubViewport
    var _viewport_b: SubViewport
    var _grid_mesh_instance: MeshInstance2D
    var _warp_material: ShaderMaterial
    var _dome: MeshInstance3D
    var _dome_material: ShaderMaterial

    # Ping-pong state
    var _read_viewport: SubViewport  # Previous frame (read from)
    var _write_viewport: SubViewport  # Current frame (render to)

    func _ready() -> void:
        call_deferred("_initialize")

    func _initialize() -> void:
        # Get scene node references
        _viewport_a = $SubViewportA
        _viewport_b = $SubViewportB
        _grid_mesh_instance = $SubViewportA/GridMeshInstance
        _warp_material = _grid_mesh_instance.material
        _dome = $WarpDome
        _dome_material = _dome.material_override

        # Create grid mesh
        var mesh := _grid_mesh.create_mesh()
        _grid_mesh_instance.mesh = mesh

        # Initialize SubViewport sizes
        _viewport_a.size = Vector2i(VIEWPORT_SIZE, VIEWPORT_SIZE)
        _viewport_b.size = Vector2i(VIEWPORT_SIZE, VIEWPORT_SIZE)

        # Set initial ping-pong state
        _read_viewport = _viewport_b
        _write_viewport = _viewport_a

        # Connect frame_post_draw for ping-pong swap
        RenderingServer.frame_post_draw.connect(_on_frame_rendered)

        # Set dome texture to read from viewport A
        call_deferred("_setup_dome_texture")

        # Load default preset
        var preset := _preset_loader.load_preset("res://presets/test_simple.milk")
        if preset:
            load_preset_obj(preset)

        print("[MilkDrop] Renderer initialized (%dx%d)" % [VIEWPORT_SIZE, VIEWPORT_SIZE])

    func _setup_dome_texture() -> void:
        _dome_material.set_shader_parameter("milkdrop_texture", _write_viewport.get_texture())

    func load_preset_obj(preset: MilkdropPreset) -> void:
        _preset = preset
        _frame_count = 0
        _per_frame_pool.clear()

        # Initialize per-frame pool with preset defaults
        _reset_per_frame_pool()

        # Run per-frame init code (once on load)
        if _preset.per_frame_init_ast != null:
            _nseel.evaluate_compiled(_preset.per_frame_init_ast, _per_frame_pool)

        print("[MilkDrop] Loaded preset: %s" % preset.name)

    func load_preset_file(path: String) -> void:
        var preset := _preset_loader.load_preset(path)
        if preset:
            load_preset_obj(preset)

    func _reset_per_frame_pool() -> void:
        # Seed per-frame pool with preset default values each frame
        _per_frame_pool["zoom"] = _preset.zoom
        _per_frame_pool["zoomexp"] = _preset.zoomexp
        _per_frame_pool["rot"] = _preset.rot
        _per_frame_pool["warp"] = _preset.warp
        _per_frame_pool["cx"] = _preset.cx
        _per_frame_pool["cy"] = _preset.cy
        _per_frame_pool["dx"] = _preset.dx
        _per_frame_pool["dy"] = _preset.dy
        _per_frame_pool["sx"] = _preset.sx
        _per_frame_pool["sy"] = _preset.sy
        _per_frame_pool["decay"] = _preset.f_decay
        _per_frame_pool["wave_r"] = _preset.wave_r
        _per_frame_pool["wave_g"] = _preset.wave_g
        _per_frame_pool["wave_b"] = _preset.wave_b
        _per_frame_pool["wave_a"] = _preset.wave_a
        _per_frame_pool["wave_x"] = _preset.wave_x
        _per_frame_pool["wave_y"] = _preset.wave_y
        _per_frame_pool["wave_mystery"] = _preset.wave_mystery

    func _process(delta: float) -> void:
        if _preset == null:
            return

        _time += delta
        _frame_count += 1

        # === MilkDrop Per-Frame Pipeline ===

        # 1. Update audio data

        var temp := _read_viewport

        _write_viewport = temp

        _dome_material.set_shader_parameter("milkdrop_texture", _read_viewport.get_texture())
    ```
