    **scripts/milkdrop/waveform_drawer.gd** -- Draws built-in waveform modes using ImmediateMesh.

    ```gdscript
    class_name MilkdropWaveformDrawer
    extends RefCounted

    # Generates audio waveform vertices for built-in modes.
    # Called by renderer each frame. Returns nothing -- draws directly into ImmediateMesh.

    func draw(im: ImmediateMesh, preset: MilkdropPreset, pool: Dictionary,
              audio_bridge: MilkdropAudioBridge, time: float) -> void:
        im.clear_surfaces()

        var mode: int = int(pool.get("wave_mode", float(preset.n_wave_mode)))
        var wave_r: float = pool.get("wave_r", preset.wave_r)
        var wave_g: float = pool.get("wave_g", preset.wave_g)
        var wave_b: float = pool.get("wave_b", preset.wave_b)
        var wave_a: float = pool.get("wave_a", preset.wave_a)
        var wave_x: float = pool.get("wave_x", preset.wave_x)
        var wave_y: float = pool.get("wave_y", preset.wave_y)
        var mystery: float = pool.get("wave_mystery", preset.wave_mystery)

        # Clamp alpha
        wave_a = clampf(wave_a, 0.0, 1.0)
        if wave_a < 0.01:
            return  # Invisible, skip drawing

        match mode:
            0: _draw_circle(im, wave_x, wave_y, mystery, wave_r, wave_g, wave_b, wave_a, audio_bridge, time)
            5: _draw_explosive(im, wave_x, wave_y, wave_r, wave_g, wave_b, wave_a, audio_bridge)
            6: _draw_line(im, wave_x, wave_y, wave_r, wave_g, wave_b, wave_a, audio_bridge, false)
            7: _draw_line(im, wave_x, wave_y, wave_r, wave_g, wave_b, wave_a, audio_bridge, true)
            _: _draw_circle(im, wave_x, wave_y, mystery, wave_r, wave_g, wave_b, wave_a, audio_bridge, time)
    ```

    Implement 4 waveform modes:

    **Mode 0 (Circle):** Circular waveform pulsing with audio. 64 vertices in a circle centered at (wave_x, wave_y). Radius = 0.4 + bass * 0.3 + mystery. Each vertex radius modulated by audio energy at that angle. Uses LINE_STRIP, close the loop by adding first vertex at end.

    **Mode 5 (Explosive Hash):** Burst pattern radiating from center. 32 line segments from center outward, length proportional to frequency band energy. Each line at evenly spaced angles. Uses individual LINE_STRIP pairs (start, end) per segment -- call surface_begin/end per segment.

    **Mode 6 (Line):** Horizontal oscilloscope line. 64 vertices across x from 0.1 to 0.9, y = wave_y + bass * 0.2 * sin(x * PI * 4 + time). Uses LINE_STRIP.

    **Mode 7 (Double Line):** Two parallel horizontal lines offset by 0.05 in y. Same as mode 6 but draw twice with slight y offset and different channel (bass for top, treb for bottom).

    Since we don't have raw PCM waveform data from AudioManager (only FFT bands), approximate waveform shapes using band magnitudes and sine functions. This gives a visually similar result. The key is that SOMETHING colorful is injected into the feedback loop each frame.

    Map all vertex positions to -1..1 space (matching the SubViewport's coordinate system for canvas_item rendering). Actually, since waveforms render as 3D overlay on top of the feedback viewport, positions should be in 0..1 UV space mapped to -1..1 screen space: x_screen = wave_x * 2.0 - 1.0.

    **scripts/milkdrop/custom_wave.gd** -- Evaluates custom waveform per-point equations.

    ```gdscript
    class_name MilkdropCustomWave
    extends RefCounted

    func draw(im: ImmediateMesh, wave_cfg, nseel: NseelInterpreter,
              per_frame_pool: Dictionary, audio_bridge: MilkdropAudioBridge,
              time: float) -> void:
        if not wave_cfg.enabled:
            return
        if wave_cfg.per_point_ast == null and wave_cfg.per_frame_ast == null:
            return

        var num_samples: int = clampi(wave_cfg.samples, 2, 512)

        # Create wave pool with q1-q32 from per-frame + t1-t8 local + audio vars
        var pool := {}
        for qi in range(1, 33):
            var qname := "q%d" % qi
            if per_frame_pool.has(qname):
                pool[qname] = per_frame_pool[qname]
        pool["time"] = time
        audio_bridge.inject_into_pool(pool)

        # Run wave init code (once per frame, sets up t1-t8 etc.)
        if wave_cfg.init_ast != null:
            nseel.evaluate_compiled(wave_cfg.init_ast, pool)

        # Run wave per-frame code
        if wave_cfg.per_frame_ast != null:
            nseel.evaluate_compiled(wave_cfg.per_frame_ast, pool)

        # Get wave colors from pool (per-frame code may have modified them)
        var r: float = pool.get("r", wave_cfg.r)
        var g: float = pool.get("g", wave_cfg.g)
        var b: float = pool.get("b", wave_cfg.b)
        var a: float = pool.get("a", wave_cfg.a)

        if a < 0.01:
            return

        # Generate points via per-point equations
        var points: PackedVector2Array = []
        points.resize(num_samples)
        for i in range(num_samples):
            var sample_pct := float(i) / float(num_samples - 1)
            pool["sample"] = sample_pct
            pool["value1"] = audio_bridge.bass * (1.0 - sample_pct) + audio_bridge.treb * sample_pct
            pool["value2"] = audio_bridge.mid

            # Default position (overridden by per-point code)
            pool["x"] = 0.5 + 0.3 * cos(sample_pct * TAU)
            pool["y"] = 0.5 + 0.3 * sin(sample_pct * TAU)

            if wave_cfg.per_point_ast != null:
                nseel.evaluate_compiled(wave_cfg.per_point_ast, pool)

            points[i] = Vector2(pool.get("x", 0.5), pool.get("y", 0.5))

        # Draw as line strip
        var prim: int = Mesh.PRIMITIVE_LINE_STRIP
        if wave_cfg.b_use_dots:
            prim = Mesh.PRIMITIVE_POINTS
        im.surface_begin(prim)
        for i in range(num_samples):
            im.surface_set_color(Color(r, g, b, a))
            im.surface_add_vertex(Vector3(points[i].x * 2.0 - 1.0, points[i].y * 2.0 - 1.0, 0.0))
        im.surface_end()
    ```

    **scripts/milkdrop/custom_shape.gd** -- Draws custom shapes as filled polygons.

    ```gdscript
    class_name MilkdropCustomShapeDrawer
    extends RefCounted

    func draw(im: ImmediateMesh, shape_cfg, nseel: NseelInterpreter,
              per_frame_pool: Dictionary, time: float) -> void:
        if not shape_cfg.enabled:
            return

        # Create shape pool with q1-q32 from per-frame + t1-t8 local
        var pool := {}
        for qi in range(1, 33):
            var qname := "q%d" % qi
            if per_frame_pool.has(qname):
                pool[qname] = per_frame_pool[qname]
        pool["time"] = time
        pool["x"] = shape_cfg.x
        pool["y"] = shape_cfg.y
        pool["rad"] = shape_cfg.rad
        pool["ang"] = shape_cfg.ang
        pool["sides"] = float(shape_cfg.sides)
        pool["r"] = shape_cfg.r; pool["g"] = shape_cfg.g
        pool["b"] = shape_cfg.b; pool["a"] = shape_cfg.a
        pool["r2"] = shape_cfg.r2; pool["g2"] = shape_cfg.g2
        pool["b2"] = shape_cfg.b2; pool["a2"] = shape_cfg.a2
        pool["border_r"] = shape_cfg.border_r; pool["border_g"] = shape_cfg.border_g
        pool["border_b"] = shape_cfg.border_b; pool["border_a"] = shape_cfg.border_a

        # Run shape init (once per load, handled by renderer)
        # Run shape per-frame code
        if shape_cfg.per_frame_ast != null:
            nseel.evaluate_compiled(shape_cfg.per_frame_ast, pool)

        var cx: float = pool.get("x", 0.5)
        var cy: float = pool.get("y", 0.5)
        var rad: float = pool.get("rad", 0.1)
        var ang: float = pool.get("ang", 0.0)
        var sides: int = clampi(int(pool.get("sides", 4.0)), 3, 100)
        var r1: float = pool.get("r", 1.0)
        var g1: float = pool.get("g", 0.0)
        var b1: float = pool.get("b", 0.0)
        var a1: float = pool.get("a", 1.0)
        var r_2: float = pool.get("r2", 0.0)
        var g_2: float = pool.get("g2", 0.0)
        var b_2: float = pool.get("b2", 0.0)
        var a_2: float = pool.get("a2", 0.0)

        if a1 < 0.01 and a_2 < 0.01:
            return

        # Draw filled polygon: triangle fan from center to N+1 perimeter vertices
        im.surface_begin(Mesh.PRIMITIVE_TRIANGLE_FAN)
        # Center vertex (color 1)
        im.surface_set_color(Color(r1, g1, b1, a1))
        im.surface_add_vertex(Vector3(cx * 2.0 - 1.0, cy * 2.0 - 1.0, 0.0))
        # Perimeter vertices (color 2)
        for i in range(sides + 1):
            var theta: float = ang + float(i) / float(sides) * TAU
            var px: float = cx + rad * cos(theta)
            var py: float = cy + rad * sin(theta)
            im.surface_set_color(Color(r_2, g_2, b_2, a_2))
            im.surface_add_vertex(Vector3(px * 2.0 - 1.0, py * 2.0 - 1.0, 0.0))
        im.surface_end()

        # Draw border outline if border_a > 0
        var ba: float = pool.get("border_a", 0.0)
        if ba > 0.01:
            var br: float = pool.get("border_r", 1.0)
            var bg: float = pool.get("border_g", 1.0)
            var bb: float = pool.get("border_b", 1.0)
            im.surface_begin(Mesh.PRIMITIVE_LINE_STRIP)
            for i in range(sides + 1):
                var theta: float = ang + float(i) / float(sides) * TAU
                var px: float = cx + rad * cos(theta)
                var py: float = cy + rad * sin(theta)
                im.surface_set_color(Color(br, bg, bb, ba))
                im.surface_add_vertex(Vector3(px * 2.0 - 1.0, py * 2.0 - 1.0, 0.0))
            im.surface_end()
    ```

    **shaders/milkdrop_blur.gdshader** -- Copy the exact shader from the research doc (MilkDrop's 8-tap weights):

    ```glsl
    shader_type canvas_item;

    uniform sampler2D source_texture : filter_linear;
    uniform vec2 texel_size;  // 1.0/texture_size
    uniform int direction;     // 0=horizontal, 1=vertical

    // MilkDrop's exact 8-tap weights (from milkdropfs.cpp BlurPasses)
    const float w[8] = float[8](4.0, 3.8, 3.5, 2.9, 1.9, 1.2, 0.7, 0.3);

    void fragment() {
        // Pair adjacent taps for efficiency (4 texture fetches instead of 8)
        float w1 = w[0] + w[1];
        float w2 = w[2] + w[3];
        float w3 = w[4] + w[5];
        float w4 = w[6] + w[7];
        float w_total = w1 + w2 + w3 + w4;

        // Offset within paired tap (weighted center)
        float d1 = 0.0 + 2.0 * w[1] / w1;
        float d2 = 2.0 + 2.0 * w[3] / w2;
        float d3 = 4.0 + 2.0 * w[5] / w3;
        float d4 = 6.0 + 2.0 * w[7] / w4;

        vec2 step_dir = (direction == 0) ? vec2(texel_size.x, 0.0) : vec2(0.0, texel_size.y);

        vec4 color = vec4(0.0);
        color += texture(source_texture, UV + step_dir * d1) * w1;
        color += texture(source_texture, UV - step_dir * d1) * w1;
        color += texture(source_texture, UV + step_dir * d2) * w2;
        color += texture(source_texture, UV - step_dir * d2) * w2;
        color += texture(source_texture, UV + step_dir * d3) * w3;
        color += texture(source_texture, UV - step_dir * d3) * w3;
        color += texture(source_texture, UV + step_dir * d4) * w4;
        color += texture(source_texture, UV - step_dir * d4) * w4;
        color /= (w_total * 2.0);

        COLOR = color;
    }
    ```

    _run_blur_passes():
    ```gdscript
    func _run_blur_passes() -> void:
        # Feed warp output through 3 levels of separable Gaussian blur.
        # Level 1: source = write viewport texture, output = blur_viewports[0] (H) then [1] (V)
        # Level 2: source = level 1 output, output = blur_viewports[2] (H) then [3] (V)
        # Level 3: source = level 2 output, output = blur_viewports[4] (H) then [5] (V)
        #
        # The blur results are available for compositing but for MVP we just use the
        # final write viewport output (warp + overlays) for the dome.
        # Blur results could be blended in later for the glow effect.
        var source_tex = _write_viewport.get_texture()
        for level in range(3):
            var h_idx := level * 2
            var v_idx := h_idx + 1
            var size := [1024, 512, 256][level]
            var texel := Vector2(1.0 / size, 1.0 / size)

            # Horizontal pass
            var h_mat: ShaderMaterial = _blur_rects[h_idx].material
            h_mat.set_shader_parameter("source_texture", source_tex)
            h_mat.set_shader_parameter("texel_size", texel)
            h_mat.set_shader_parameter("direction", 0)

            # Vertical pass
            var v_mat: ShaderMaterial = _blur_rects[v_idx].material
            v_mat.set_shader_parameter("source_texture", _blur_viewports[h_idx].get_texture())
            v_mat.set_shader_parameter("texel_size", texel)
            v_mat.set_shader_parameter("direction", 1)

            source_tex = _blur_viewports[v_idx].get_texture()
    ```
