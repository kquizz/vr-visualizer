    Create `scripts/milkdrop/preset_loader.gd`:

    ```gdscript
    class_name MilkdropPresetLoader
    extends RefCounted

    var _nseel: NseelInterpreter  # Set in constructor or create new

    func _init():
        _nseel = NseelInterpreter.new()

    func load_preset(path: String) -> MilkdropPreset:
        # 1. Read file as text
        var file := FileAccess.open(path, FileAccess.READ)
        if file == null:
            push_error("PresetLoader: Cannot open %s" % path)
            return null
        var text := file.get_as_text()
        file.close()

        # 2. Parse into key-value pairs
        var preset := MilkdropPreset.new()
        preset.name = path.get_file().get_basename()
        var kv := _parse_ini(text)

        # 3. Map scalar parameters
        _apply_scalars(preset, kv)

        # 4. Concatenate numbered equation lines
        preset.per_frame_init_code = _concat_numbered(kv, "per_frame_init_")
        preset.per_frame_code = _concat_numbered(kv, "per_frame_")
        preset.per_pixel_code = _concat_numbered(kv, "per_pixel_")

        # 5. Parse custom waves (wavecode_0..3, wave_0_init, wave_0_per_frame, wave_0_per_point)
        for i in range(4):
            var wave := _parse_custom_wave(kv, i)
            preset.custom_waves.append(wave)

        # 6. Parse custom shapes (shapecode_0..3, shape_0_init, shape_0_per_frame)
        for i in range(4):
            var shape := _parse_custom_shape(kv, i)
            preset.custom_shapes.append(shape)

        # 7. Compile all equations into ASTs
        _compile_equations(preset)

        return preset
    ```

    Key implementation details:

    **_parse_ini(text):** Split text into lines. Skip lines before [preset00]. For each line after [preset00], split on FIRST "=" only (so "per_frame_1=zoom = zoom + 0.1" splits to key="per_frame_1", value="zoom = zoom + 0.1"). Strip comments (// to end of line). Store in Dictionary[String, String].

    **_concat_numbered(kv, prefix):** Find all keys matching prefix + number (per_frame_1, per_frame_2, ...). Sort numerically. Join values with ";". Skip keys that also match other prefixes -- e.g., when looking for "per_frame_" do NOT match "per_frame_init_". Use exact matching: key must start with prefix followed by a digit.

    **_apply_scalars(preset, kv):** Map known keys to preset fields. Use a mapping Dictionary:
    ```
    "fDecay" -> "f_decay", "fGammaAdj" -> "f_gamma_adj",
    "zoom" -> "zoom", "rot" -> "rot", "warp" -> "warp",
    "cx" -> "cx", "cy" -> "cy", "dx" -> "dx", "dy" -> "dy",
    "sx" -> "sx", "sy" -> "sy", "zoomexp" -> "zoomexp",
    "fWarpAnimSpeed" -> "warp_anim_speed", "fWarpScale" -> "warp_scale",
    "nWaveMode" -> int, "bAdditiveWaves" -> bool, "bWaveDots" -> bool, "bWaveThick" -> bool,
    "wave_r", "wave_g", "wave_b", "wave_a", "wave_x", "wave_y", "wave_mystery",
    "fWaveAlpha", "fWaveScale", "fWaveSmoothing",
    "bDarkenCenter", "bInvert", "bBrighten", "bSolarize",
    "bMaximizeWaveColor" -> bool
    ```
    For bool fields: value "1" = true, "0" = false.
    For int fields: value.to_int().
    For float fields: value.to_float().

    **_parse_custom_wave(kv, index):** Read wavecode_{index}_enabled, wavecode_{index}_samples, etc. Read wave_{index}_init, wave_{index}_per_frame, wave_{index}_per_point (numbered lines, same concatenation).

    **_parse_custom_shape(kv, index):** Read shapecode_{index}_enabled, shapecode_{index}_sides, shapecode_{index}_x, etc. Read shape_{index}_init, shape_{index}_per_frame (numbered lines).

    **_compile_equations(preset):** For each non-empty equation string, call _nseel.compile(code) and store result in the corresponding _ast field. Wrap in try/push_warning for malformed equations (graceful skip per CONTEXT.md).
