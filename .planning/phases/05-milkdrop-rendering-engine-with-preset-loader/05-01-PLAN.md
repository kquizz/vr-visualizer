       ```cpp
       #include "register_types.h"
       #include "projectm_wrapper.h"

       #include <gdextension_interface.h>
       #include <godot_cpp/core/defs.hpp>
       #include <godot_cpp/godot.hpp>

       using namespace godot;

       void initialize_projectm_module(ModuleInitializationLevel p_level) {
           if (p_level != MODULE_INITIALIZATION_LEVEL_SCENE) {
               return;
           }
           ClassDB::register_class<ProjectMWrapper>();
       }

       void uninitialize_projectm_module(ModuleInitializationLevel p_level) {
           if (p_level != MODULE_INITIALIZATION_LEVEL_SCENE) {
               return;
           }
       }

       extern "C" {
       GDExtensionBool GDE_EXPORT projectm_library_init(
           GDExtensionInterfaceGetProcAddress p_get_proc_address,
           const GDExtensionClassLibraryPtr p_library,
           GDExtensionInitialization *r_initialization) {
           godot::GDExtensionBinding::InitObject init_obj(p_get_proc_address, p_library, r_initialization);
           init_obj.register_initializer(initialize_projectm_module);
           init_obj.register_terminator(uninitialize_projectm_module);
           init_obj.set_minimum_library_initialization_level(MODULE_INITIALIZATION_LEVEL_SCENE);
           return init_obj.init();
       }
       }
       ```

    1. Create `addons/projectm/src/projectm_wrapper.h`:
       ```cpp
       #ifndef PROJECTM_WRAPPER_H
       #define PROJECTM_WRAPPER_H

       #include <godot_cpp/classes/node.hpp>
       #include <godot_cpp/classes/image.hpp>
       #include <godot_cpp/classes/image_texture.hpp>
       #include <godot_cpp/variant/packed_float32_array.hpp>

       #include <projectM-4/projectM.h>

       namespace godot {

       class ProjectMWrapper : public Node {
           GDCLASS(ProjectMWrapper, Node)

       private:
           projectm_handle _handle = nullptr;
           bool _initialized = false;
           int _width = 1024;
           int _height = 1024;
           Ref<ImageTexture> _texture;
           PackedByteArray _pixel_buffer;

       protected:
           static void _bind_methods();

       public:
           ProjectMWrapper();
           ~ProjectMWrapper();

           // Core API exposed to GDScript
           bool initialize(int width, int height);
           void shutdown();
           bool is_initialized() const;

           // Preset management
           bool load_preset(const String &path);
           String get_preset_name() const;

           // Audio input (raw PCM float samples, interleaved stereo)
           void feed_audio(const PackedFloat32Array &samples);

           // Rendering
           void render_frame();
           Ref<ImageTexture> get_texture() const;

           // Configuration
           void set_viewport_size(int width, int height);
       };

       } // namespace godot

       #endif
       ```

    2. Create `addons/projectm/src/projectm_wrapper.cpp`:
       - `initialize(width, height)`: Call `projectm_create()`, then `projectm_set_window_size()`. Set `_initialized = true`. Allocate `_pixel_buffer` as `PackedByteArray` of size `width * height * 4` (RGBA). Create `_texture` as `ImageTexture`.
       - `shutdown()`: Call `projectm_destroy(_handle)`, set `_handle = nullptr`, `_initialized = false`.
       - `load_preset(path)`: Convert Godot `String` path to `const char*`. If path starts with "res://", use `ProjectSettings::get_singleton()->globalize_path(path)` to get absolute path. Call `projectm_load_preset_file(_handle, abs_path.utf8().get_data(), false)`. Return true on success.
       - `feed_audio(samples)`: Call `projectm_pcm_add_float(_handle, samples.ptr(), samples.size() / 2, PROJECTM_STEREO)`. If samples is mono (odd count), use `PROJECTM_MONO`.
       - `render_frame()`: Call `projectm_opengl_render_frame(_handle)`. Then read pixels via `glReadPixels(0, 0, _width, _height, GL_RGBA, GL_UNSIGNED_BYTE, _pixel_buffer.ptrw())`. Create Image from `_pixel_buffer` with `Image::create_from_data(_width, _height, false, Image::FORMAT_RGBA8, _pixel_buffer)`. Update `_texture` via `_texture->set_image(image)`.
       - `get_texture()`: Return `_texture`.
       - `_bind_methods()`: Bind all public methods with `ClassDB::bind_method(D_METHOD(...), &ProjectMWrapper::...)`.
       - Constructor: Initialize all members to defaults.
       - Destructor: Call `shutdown()` if initialized.

       Important implementation notes:
       - Include `<GL/gl.h>` (or `<OpenGL/gl.h>` on macOS) for glReadPixels
       - The glReadPixels approach is the simplest texture transfer. If projectM provides `projectm_get_texture_id()`, we could use shared GL context, but pixel readback is reliable and sufficient for 1024x1024 at 90fps.
       - Handle errors gracefully: if `projectm_create()` returns null, log error and return false from initialize().
       - Use `WARN_PRINT` and `ERR_PRINT` macros from godot-cpp for logging.

    3. Build the extension:
       ```
       cd addons/projectm
       scons platform=macos
       ```

    4. If build fails, debug and fix. Common issues:

       - godot-cpp API changes: check godot-cpp docs for 4.x class registration syntax
       - OpenGL headers on macOS: use `#include <OpenGL/gl.h>` not `<GL/gl.h>`
